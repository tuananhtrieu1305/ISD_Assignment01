import React,{useEffect,useRef,useState} from 'react';
import {createRoot} from 'react-dom/client';
import './style.css';
import './comparison.css';

const API=(import.meta.env.VITE_API_BASE_URL||'').replace(/\/$/,'');
const BRANCHES=[['llm_direct','LLM trực tiếp','Kiến thức sẵn có của Gemini'],['agentic_rag','Đa agent + RAG','Truy hồi · KG · kiểm chứng']];
const fmt=(v,d=1)=>v==null?'—':Number(v).toLocaleString('vi-VN',{maximumFractionDigits:d});
const stateName={complete:'Hoàn tất',verification_failed:'Chưa kiểm chứng đạt',provider_error:'Lỗi provider',waiting_quota:'Chờ quota',running:'Đang xử lý',queued:'Đang chờ',failed:'Chưa hoàn tất',partial:'Hoàn tất một phần'};
async function api(path,body){const response=await fetch(API+path,body?{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(body)}:{});const data=await response.json();if(!response.ok)throw new Error(data.error?.message||`HTTP ${response.status}`);return data;}
function safeUrl(value){try{const u=new URL(value);return ['http:','https:'].includes(u.protocol)?u.href:null;}catch{return null;}}

function App(){
 const [tab,setTab]=useState('input'),[ready,setReady]=useState(null),[examples,setExamples]=useState(null),[summary,setSummary]=useState(null);
 const [question,setQuestion]=useState('Vận động có ích gì cho cân nặng và đường huyết?'),[job,setJob]=useState(null),[error,setError]=useState(''),[busy,setBusy]=useState(false);
 const [profile,setProfile]=useState({}),[risk,setRisk]=useState(null),[useProfile,setUseProfile]=useState(false),[saved,setSaved]=useState([]);
 const timer=useRef(null);
 const refresh=()=>fetch(API+'/ready').then(r=>r.json()).then(setReady).catch(()=>setReady({status:'unavailable'}));
 useEffect(()=>{refresh();api('/api/v2/examples').then(data=>{setExamples(data);setProfile(data.profiles[0].profile);}).catch(e=>setError(e.message));api('/api/v2/summary').then(setSummary).catch(()=>{});api('/api/v2/development').then(r=>setSaved(r.rows)).catch(()=>{});return()=>clearTimeout(timer.current);},[]);
 async function poll(id){try{const next=await api('/api/v2/jobs/'+id);setJob(next);if(['queued','running'].includes(next.status))timer.current=setTimeout(()=>poll(id),2000);else{setBusy(false);refresh();}}catch(e){setError(e.message);setBusy(false);}}
 async function run(event){event.preventDefault();if(busy)return;clearTimeout(timer.current);setError('');setBusy(true);setJob(null);setTab('result');try{const data=await api('/api/v2/compare',{question,mode:useProfile?'combined':'knowledge',profile:useProfile?profile:null});poll(data.id);}catch(e){setError(e.message);setBusy(false);}}
 async function resume(){if(!job?.id)return;setBusy(true);setError('');try{await api('/api/v2/jobs/'+job.id+'/resume',{});poll(job.id);}catch(e){setError(e.message);setBusy(false);}}
 async function predict(){setError('');try{setRisk(await api('/api/v2/risk',{profile}));}catch(e){setError(e.message);}}
 const results=job?.results||{};
 function showSaved(row){clearTimeout(timer.current);setBusy(false);setJob({...row,status:'complete',saved:true});setTab('result');}
 return <div className="app"><header className="header"><a className="brand" href="#" onClick={e=>{e.preventDefault();setTab('input');}}><span className="brand-icon">D<span>↗</span></span><span><small>PROJECT 01 · TRÍ TUỆ NHÂN TẠO</small><strong>Diabetes <em>/</em> Agent Lab</strong></span></a><span className={`health ${ready?.status==='ready'?'ready':'unavailable'}`}><i/>{ready?.status==='ready'?'Gemini + dữ liệu sẵn sàng':'Đang kiểm tra / chuẩn bị dữ liệu'}</span></header>

Không dùng để chẩn đoán, kê đơn hoặc thay thế tư vấn chuyên môn.</small></div><div className="quota-mini"><b>{ready?.usage?.generation_attempts??'—'} / 18</b><span>request đã ghi nhận trong ngày provider<br/>Free tier giới hạn 20/ngày; app chừa 2.</span></div></aside>
 <main><div className="page-heading"><span className="eyebrow">CÙNG MODEL · KHÁC NGUỒN THÔNG TIN</span><h1>{tab==='input'?'Hỏi rõ. Hiểu có căn cứ.':tab==='result'?'Hai câu trả lời, cạnh nhau.':'Đọc kết quả một cách công bằng.'}</h1><p>{tab==='input'?'Đặt Gemini trực tiếp cạnh hệ đa agent dùng truy hồi, đồ thị tri thức và kiểm chứng.':tab==='result'?'Đọc câu trả lời trước. Mở nguồn và quá trình xử lý khi bạn cần đối chiếu.':'Tách lượt demo hiện tại khỏi bộ đánh giá nhiều ngày. Không mặc định RAG tốt hơn.'}</p></div>
 {error&&<div className="alert error" role="alert">{error}</div>}
 {tab==='input'&&<form onSubmit={run}><section className="panel prompt-panel"><span className="eyebrow">CÂU HỎI KIẾN THỨC</span><h2>Bạn muốn tìm hiểu điều gì?</h2><label className="field question"><span>Câu hỏi của bạn</span><textarea value={question} maxLength={4096} rows={4} onChange={e=>setQuestion(e.target.value)} required disabled={busy}/></label><div className="examples"><span>THỬ NHANH</span>{examples?.questions.map(e=><button key={e.title} type="button" disabled={busy} onClick={()=>setQuestion(e.question)}>{e.title} ↗</button>)}</div><div className="comparison-explainer"><div><b>01 / LLM trực tiếp</b><p>Dùng kiến thức Gemini, không tra kho project.</p></div><div><b>02 / Đa agent + RAG</b><p>Viết truy vấn → tìm nguồn + KG → xếp hạng → trả lời → kiểm chứng.</p></div></div><button type="submit" className="run-button" disabled={busy||ready?.status!=='ready'||!ready?.live_enabled}>{busy?'Đang xử lý…':'So sánh hai câu trả lời'}<span>→</span></button><p className="footnote">Một lượt thường dùng5 request Gemini và có thể mất vài phút.  <details className="panel optional-risk"><summary>Hồ sơ CDC · demo công cụ bổ sung</summary><p>21 field khảo sát, chỉ tính bằng checkpoint cố định; không gửi hồ sơ lên Gemini. Score là đầu ra nhãn khảo sát lịch sử, không là xác suất mắc mới.</p><label><input type="checkbox" checked={useProfile} onChange={e=>setUseProfile(e.target.checked)}/> Kèm Risk Agent trong lượt RAG (câu trả lời kiến thức vẫn là so sánh chính)</label><div className="profile-grid">{Object.entries(examples?.schema?.limits||{}).map(([name,bounds])=><label className="field" key={name}><span>{name==='bmi'?'BMI':name}</span><input aria-label={name==='bmi'?'BMI':name} type="number" min={bounds[0]} max={bounds[1]} step={name==='bmi'?'.1':'1'} value={profile[name]??''} onChange={e=>setProfile({...profile,[name]:e.target.value===''?null:Number(e.target.value)})}/></label>)}</div><button type="button" onClick={predict}>Tính score CDC · không dùng API</button>{risk&&<p role="status">Score: <b>{fmt(risk.score,6)}</b> · {risk.class_name}.  {tab==='result'&&<>{job?<><section className="submitted"><span className="eyebrow">CÂU HỎI ĐÃ GỬI</span><h2>{job.question}</h2><div className="run-meta"><span className="tag">{job.saved?'Development đã lưu · không phải lượt live mới':stateName[job.status]||job.status}</span>{busy&&<span role="status" className="live-progress"><span className="spinner"/>{job.progress||'Đang khởi tạo…'}</span>}</div>{job.error&&<div className="alert" role="status">{job.error.message}{job.status==='waiting_quota'&&<button disabled={busy} onClick={resume}>Tiếp tục lượt này</button>}</div>}</section><div className="answer-comparison">{BRANCHES.map(([id,title,subtitle])=><Answer key={id} id={id} title={title} subtitle={subtitle} result={results[id]} busy={busy}/>)}</div>{results.agentic_rag&&<Evidence result={results.agentic_rag}/>}</>:<section className="panel empty"><h2>Chưa có câu trả lời.</h2><p>Gửi một câu hỏi hoặc xem development đã lưu.</p><button onClick={()=>setTab('input')}>Về câu hỏi →</button></section>}</>}
 {tab==='compare'&&<><section className="panel"><span className="eyebrow">LƯỢT HIỆN TẠI</span><h2>Đo toàn bộ quá trình</h2><div className="table-scroll"><table><thead><tr><th>Nhánh</th><th>Trạng thái</th><th>Thời gian</th><th>Call / cache</th><th>Chi phí cấu hình</th></tr></thead><tbody>{BRANCHES.map(([id,title])=>{const r=results[id];return <tr key={id}><td>{title}</td><td>{r?stateName[r.status]:'Chưa chạy'}</td><td>{fmt(r?.latency_ms/1000)} s</td><td>{r?.usage?.filter(x=>x.cache_status==='live').length??'—'} / {r?.usage?.filter(x=>x.cache_status==='local_cache').length??'—'}</td><td>{r?'0 USD · free tier':'—'}</td></tr>;})}</tbody></table></div><p className="footnote">Call gồm planner/embedding/rerank/sinh/kiểm chứng. Cache được ghi riêng. Chưa kiểm hóa đơn; không lấy một lượt demo làm kết luận chất lượng.</p></section><section className="panel"><span className="eyebrow">HELD-OUT MỚI · 40 CÂU × 2 NHÁNH</span><h2>{summary?.status==='outputs_complete_review_pending'?'Đủ đầu ra, đang chờ chấm':'Đánh giá chưa hoàn tất'}</h2><p>Chỉ chạy sau development/freeze. Quota20 request/ngày nên chia nhiều ngày; web không tự mở benchmark.</p><div className="table-scroll"><table><thead><tr><th>Nhánh</th><th>Đầu ra /40</th><th>Hoàn tất</th><th>Lỗi</th><th>Chất lượng</th></tr></thead><tbody>{summary?.rows?.map(r=><tr key={r.variant}><td>{r.variant}</td><td>{r.outputs}/40</td><td>{r.complete}</td><td>{r.errors}</td><td>Chờ review</td></tr>)}</tbody></table></div><p className="footnote">P0/P1/P2 và82,5% thuộc bản offline lịch sử, không phải chất lượng hệ LLM này.</p></section></>}
 <footer>PROJECT01 <span>·</span> Gemini + MedQuAD + CDC <span>·</span> Tiếng Việt · nguồn có thể đối chiếu</footer></main></div></div>;
}
function Answer({id,title,subtitle,result,busy}){const usage=result?.usage||[];const total=usage.reduce((s,r)=>s+(r.usage?.total_tokens||0),0);return <section className={'answer-column '+(id==='agentic_rag'?'grounded':'')} aria-label={title}><div className="answer-column-head"><span className="eyebrow">{subtitle}</span><h2>{title}</h2><small>Gemini 3.8 Flash</small></div>{result?<><span className="tag">{stateName[result.status]||result.status}</span><div className="primary-answer" style={{whiteSpace:'pre-wrap'}}>{result.answer_vi||'Không có câu trả lời hoàn chỉnh.'}</div><div className="answer-stats"><span>{fmt(result.latency_ms/1000)} s</span><span>{total?fmt(total,0)+' tokens provider':'Token chưa có usage đầy đủ'}</span><span>{result.citations?.length||0} nguồn</span></div>{result.risk&&<details><summary>Đầu ra Risk Agent · không do LLM tạo</summary><p>Score {fmt(result.risk.score,6)} · {result.risk.class_name} · không là xác suất mắc mới.</p></details>}</>:<div className="answer-placeholder"><span>{busy?'Đang thực hiện…':'Chưa có kết quả'}</span><p>{id==='agentic_rag'?'Câu trả lời xuất hiện sau khi hoàn thành bước kiểm chứng.':'Câu trả lời dùng kiến thức sẵn có của cùng model.'}</p></div>}</section>;}
